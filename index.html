function renderMenuSchedule() { const now = new Date(); const menitSekarang = (now.getHours() * 60) + now.getMinutes(); // Kelas X: 07.10 (430 menit) s/d 10.00 (600 menit) const kXStart = (7 * 60) + 10; const kXEnd = (10 * 60) + 0; // Kelas XI: 10.30 (630 menit) s/d 13.50 (830 menit) const kXIStart = (10 * 60) + 30; const kXIEnd = (13 * 60) + 50; const secX = document.getElementById('kelasXSection'); const secXI = document.getElementById('kelasXISection'); const secNo = document.getElementById('noScheduleSection'); secX.style.display = 'none'; secXI.style.display = 'none'; secNo.style.display = 'none'; if (menitSekarang >= kXStart && menitSekarang <= kXEnd) { secX.style.display = 'block'; } else if (menitSekarang >= kXIStart && menitSekarang <= kXIEnd) { secXI.style.display = 'block'; } else { secNo.style.display = 'block'; } }
